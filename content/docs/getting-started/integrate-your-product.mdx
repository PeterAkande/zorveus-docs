This guide walks through the complete end-to-end backend integration: mapping customer tiers, making attributed requests, checking balances, and granting credits upon payment.

## 1. Configure your plan keys

In the [Zorveus dashboard](https://app.zorveus.com), configure separate connections for each subscription tier your product offers:

- **Free Tier connection**: Set a $2/month connection cap, allow lightweight models (e.g. `openai/gpt-4.1-mini`), and set credit mode to `enforce`.
- **Pro Tier connection**: Set a $20/month connection cap, allow frontier models (e.g. `anthropic/claude-3-5-sonnet`), and set credit mode to `enforce`.

Store your inference keys and your service key in your backend secrets:

```bash
ZORVEUS_FREE_KEY="zrv_free_tier_key..."
ZORVEUS_PRO_KEY="zrv_pro_tier_key..."
ZORVEUS_SERVICE_KEY="zrv_svc_your_service_key..."
ZORVEUS_APP_ID="app_123"
```

## 2. Complete backend service implementation

Here is a complete, copyable backend module in Python and TypeScript that ties together plan selection, user attribution, allowance checks, and error handling.

<CodeGroup titles={["TypeScript", "Python"]}>

```typescript
import OpenAI, { APIError } from "openai";
import { ZorveusServiceClient } from "@zorveus/sdk";

const service = new ZorveusServiceClient({
  apiKey: process.env.ZORVEUS_SERVICE_KEY!,
});

const planClients = {
  free: new OpenAI({
    apiKey: process.env.ZORVEUS_FREE_KEY!,
    baseURL: "https://api.zorveus.com/v1",
  }),
  pro: new OpenAI({
    apiKey: process.env.ZORVEUS_PRO_KEY!,
    baseURL: "https://api.zorveus.com/v1",
  }),
};

interface CompletionRequest {
  userId: string;
  userTier: "free" | "pro";
  prompt: string;
}

export async function runUserPrompt({ userId, userTier, prompt }: CompletionRequest) {
  // 1. Optional pre-flight check: Verify customer has remaining allowance
  const user = await service.productUsers.getByExternalId({
    appId: process.env.ZORVEUS_APP_ID!,
    externalUserId: userId,
    upsertIfMissing: true,
  });

  const available = parseFloat(user.allowance?.available_allowance || "0");
  if (available <= 0) {
    return {
      status: "quota_exhausted",
      message: "You have reached your monthly AI allowance. Please upgrade or buy credits.",
    };
  }

  // 2. Select client for the user's tier
  const client = planClients[userTier];

  try {
    // 3. Send request with customer attribution header
    const response = await client.chat.completions.create(
      {
        model: userTier === "pro" ? "anthropic/claude-3-5-sonnet" : "openai/gpt-4.1-mini",
        messages: [{ role: "user", content: prompt }],
      },
      {
        headers: { "X-Zorveus-External-User-ID": userId },
      }
    );

    return {
      status: "success",
      content: response.choices[0].message.content,
    };
  } catch (err: unknown) {
    if (err instanceof APIError) {
      if (err.status === 403) {
        return {
          status: "quota_exhausted",
          message: "You have reached your monthly AI allowance. Please upgrade or buy credits.",
        };
      }
      if (err.status === 402) {
        return {
          status: "system_error",
          message: "AI service temporarily unavailable. Organization wallet requires funding.",
        };
      }
    }
    throw err;
  }
}
```

```python
import os
from openai import OpenAI, APIStatusError
from zorveus import ZorveusServiceClient

service = ZorveusServiceClient(api_key=os.environ["ZORVEUS_SERVICE_KEY"])
app_id = os.environ["ZORVEUS_APP_ID"]

plan_clients = {
    "free": OpenAI(api_key=os.environ["ZORVEUS_FREE_KEY"], base_url="https://api.zorveus.com/v1"),
    "pro": OpenAI(api_key=os.environ["ZORVEUS_PRO_KEY"], base_url="https://api.zorveus.com/v1"),
}

def run_user_prompt(user_id: str, user_tier: str, prompt: str):
    # 1. Optional pre-flight check: Verify customer has remaining allowance
    user = service.product_users.get_by_external_id(
        app_id=app_id,
        external_user_id=user_id,
        upsert_if_missing=True,
    )

    available = float(user.allowance.available_allowance if user.allowance else 0)
    if available <= 0:
        return {
            "status": "quota_exhausted",
            "message": "You have reached your monthly AI allowance. Please upgrade or buy credits.",
        }

    # 2. Select client for the user's tier
    client = plan_clients.get(user_tier, plan_clients["free"])
    model = "anthropic/claude-3-5-sonnet" if user_tier == "pro" else "openai/gpt-4.1-mini"

    try:
        # 3. Send request with customer attribution header
        response = client.chat.completions.create(
            model=model,
            messages=[{"role": "user", "content": prompt}],
            extra_headers={"X-Zorveus-External-User-ID": user_id},
        )
        return {
            "status": "success",
            "content": response.choices[0].message.content,
        }
    except APIStatusError as err:
        if err.status_code == 403:
            return {
                "status": "quota_exhausted",
                "message": "You have reached your monthly AI allowance. Please upgrade or buy credits.",
            }
        if err.status_code == 402:
            return {
                "status": "system_error",
                "message": "AI service temporarily unavailable. Organization wallet requires funding.",
            }
        raise err
```

## 3. Selling credit top-ups with Stripe

When customers run out of their monthly plan allowance, you can let them purchase one-time credit top-ups.

When your Stripe webhook receives a `payment_intent.succeeded` event, use your Zorveus service key to grant credits. Passing Stripe's payment intent ID as `external_reference` guarantees idempotency:

<CodeGroup titles={["TypeScript", "Python"]}>

```typescript
import { ZorveusServiceClient } from "@zorveus/sdk";
import type Stripe from "stripe";

const service = new ZorveusServiceClient({ apiKey: process.env.ZORVEUS_SERVICE_KEY! });

export async function handleStripePaymentSuccess(paymentIntent: Stripe.PaymentIntent) {
  const userId = paymentIntent.metadata.userId;
  const packAmountUsd = (paymentIntent.amount / 100).toFixed(2); // e.g. "10.00"

  // Idempotent grant: retries with the same reference return the existing receipt
  const receipt = await service.productUsers.grantCreditByExternalId({
    appId: process.env.ZORVEUS_APP_ID!,
    externalUserId: userId,
    amount: packAmountUsd,
    externalReference: paymentIntent.id,
    source: "purchase",
    reason: `Stripe purchase ${paymentIntent.id}`,
  });

  console.log(`Granted $${receipt.credit_grant.amount} to user ${userId}`);
}
```

```python
from zorveus import ZorveusServiceClient

service = ZorveusServiceClient(api_key=os.environ["ZORVEUS_SERVICE_KEY"])
app_id = os.environ["ZORVEUS_APP_ID"]

def handle_stripe_payment_success(payment_intent):
    user_id = payment_intent["metadata"]["user_id"]
    pack_amount_usd = f"{payment_intent['amount'] / 100:.2f}"

    receipt = service.product_users.grant_credit_by_external_id(
        app_id=app_id,
        external_user_id=user_id,
        amount=pack_amount_usd,
        external_reference=payment_intent["id"],
        source="purchase",
        reason=f"Stripe purchase {payment_intent['id']}",
    )

    print(f"Granted ${receipt.credit_grant.amount} to user {user_id}")
```

## 4. Key security and best practices

- **Never share inference keys with frontends**: All requests to `https://api.zorveus.com` that use your business keys must come from your authenticated application backend.
- **Always attach the external user ID**: The `X-Zorveus-External-User-ID` header is what connects an AI request to a customer's allowance and usage ledger.
- **Match billing cycles with allowance anchors**: If a user subscribes on the 15th of the month, pass `allowance_starts_at` when creating their profile so their monthly cap resets on the 15th, rather than the 1st. See [Allowance periods](/startups/allowance-periods).
