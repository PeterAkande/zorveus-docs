If you build developer tools, IDE extensions, or SaaS platforms where users bring their own AI funding, you can use **Zorveus OAuth 2.0 with PKCE (Proof Key for Code Exchange)**.

Connecting a user's Zorveus wallet enables end-users to authenticate via OAuth 2.0 PKCE and fund AI inference with their own credits.

---

## 1. App Registration

1. In [app.zorveus.com](https://app.zorveus.com), create or select a **Developer Organization**.
2. Create an application and specify your registered **Redirect URIs** (e.g. `https://myapp.com/oauth/callback`).
3. Note your **Client ID** (`zrv_client_...`) and initial **Client Secret** (`zrv_client_secret_...`).

---

## 2. React Applications (`@zorveus/react`)

This is the fastest integration method for React and Next.js applications.

### Step 1: Install dependencies

```bash
npm install @zorveus/react @zorveus/sdk
```

### Step 2: Wrap your app in `ZorveusProvider`

Provide your `clientId`, `redirectUri`, and optional `appId` in your root app wrapper:

```tsx
import React from "react";
import { ZorveusProvider } from "@zorveus/react";

export function AppProviders({ children }: { children: React.ReactNode }) {
  return (
    <ZorveusProvider
      clientId={process.env.NEXT_PUBLIC_ZORVEUS_CLIENT_ID}
      redirectUri="http://localhost:3000/oauth/callback"
      appId={process.env.NEXT_PUBLIC_ZORVEUS_APP_ID}
    >
      {children}
    </ZorveusProvider>
  );
}
```

### Step 3: Add the `<ConnectWalletButton />` component

Render the pre-built wallet connection button in your top navigation or header:

```tsx
import { ConnectWalletButton, useZorveusAuth } from "@zorveus/react";

export function Header() {
  const { isConnected } = useZorveusAuth();

  return (
    <header style={{ display: "flex", justifyContent: "space-between", padding: "16px" }}>
      <h1>My AI SaaS App</h1>
      <ConnectWalletButton
        label="Connect Zorveus Wallet"
        connectedLabel="Wallet Connected"
      />
    </header>
  );
}
```

### Step 4: Execute AI inference with the connected wallet

Use `useZorveusInference` or `useZorveusAuth` to run chat completions using the connected user's wallet access token:

```tsx
import React, { useState } from "react";
import { useZorveusInference, useZorveusAuth } from "@zorveus/react";

export function ChatBox() {
  const { isConnected } = useZorveusAuth();
  const { client, isReady } = useZorveusInference();
  const [output, setOutput] = useState("");

  if (!isConnected) {
    return <p>Please connect your Zorveus wallet above to start generating AI responses.</p>;
  }

  async function handleGenerate() {
    if (!isReady || !client) return;

    const stream = await client.chat.completions.create({
      model: "openai/gpt-4.1-mini",
      messages: [{ role: "user", content: "Write a short product pitch." }],
      stream: true,
    });

    for await (const chunk of stream) {
      const text = chunk.choices[0]?.delta?.content || "";
      setOutput((prev) => prev + text);
    }
  }

  return (
    <div>
      <button onClick={handleGenerate}>Generate</button>
      <pre>{output}</pre>
    </div>
  );
}
```

## 3. Python Applications (`zorveus`)

### Step 1: Generate PKCE parameters and redirect user

When the user clicks **"Connect Zorveus Wallet"**, generate the PKCE challenge and state in your backend, store `code_verifier` and `state` in the user's session, and redirect the user to the authorization URL:

```python
from zorveus import ZorveusOAuth

# 1. Generate PKCE values
pkce = ZorveusOAuth.generate_pkce()

# Save pkce.code_verifier and pkce.state in user's HTTP session / cookie
session["code_verifier"] = pkce.code_verifier
session["oauth_state"] = pkce.state

# 2. Build authorization URL
auth_url = ZorveusOAuth.get_authorization_url(
    client_id="zrv_client_your_client_id",
    redirect_uri="https://myapp.com/oauth/callback",
    state=pkce.state,
    code_challenge=pkce.code_challenge,
    scopes=["inference:write", "models:*"],
)

# Redirect user browser to auth_url
```

<Callout type="warning" title="Model Scope Required">
  OAuth authorization requests require at least one model scope such as `models:*` alongside `inference:write`. Omitting model scopes triggers a `zorveus_model_scope_required` authorization error.
</Callout>

### Step 2: Handle the OAuth callback

When the user authorizes your app on Zorveus, they are redirected back to your `redirect_uri` with query parameters. Validate the callback:

```python
from zorveus import ZorveusOAuth

# Read query string and saved session state
full_callback_url = request.url
saved_state = session.get("oauth_state")

validation = ZorveusOAuth.validate_callback(
    full_callback_url,
    expected_state=saved_state
)

if not validation.valid:
    raise Exception(f"OAuth validation failed: {validation.error_description}")

auth_code = validation.code
```

### Step 3: Exchange code for user access token

Exchange the authorization code and stored `code_verifier` for the user's `access_token`:

```python
saved_verifier = session.get("code_verifier")

token_data = ZorveusOAuth.exchange_token(
    client_id="zrv_client_your_client_id",
    client_secret="zrv_client_secret_your_client_secret",
    code=auth_code,
    code_verifier=saved_verifier,
    redirect_uri="https://myapp.com/oauth/callback"
)

# Save user_access_token in database associated with this user
user_access_token = token_data.access_token
```

### Step 4: Run inference billed to user wallet

Pass the user's `user_access_token` as the `api_key` to `Zorveus` or `ZorveusOpenAI`. Inference usage is billed directly to the user's connected wallet:

<CodeGroup titles={["Native Zorveus Client", "OpenAI SDK Adapter (zorveus[openai])"]}>

```python
from zorveus import Zorveus

# Instantiate client with user's access token
user_client = Zorveus(api_key=user_access_token)

response = user_client.chat.completions.create(
    model="openai/gpt-4.1-mini",
    messages=[{"role": "user", "content": "Hello!"}]
)

print(response.choices[0].message.content)
```

```python
# pip install zorveus[openai]
from zorveus.openai import ZorveusOpenAI

# Instantiate OpenAI adapter with user's access token
user_client = ZorveusOpenAI(api_key=user_access_token)

response = user_client.chat.completions.create(
    model="openai/gpt-4.1-mini",
    messages=[{"role": "user", "content": "Hello!"}]
)

print(response.choices[0].message.content)
```

## 4. Vanilla JavaScript and Node.js (`@zorveus/sdk`)

### Step 1: Initiate OAuth PKCE flow

```typescript
import { ZorveusOAuth } from "@zorveus/sdk";

export function initiateWalletConnect() {
  // 1. Generate PKCE verifier, challenge (S256), and state CSRF token
  const pkce = ZorveusOAuth.generatePKCE();

  // 2. Save verifier and state in session or localStorage for callback verification
  sessionStorage.setItem("zorveus_code_verifier", pkce.codeVerifier);
  sessionStorage.setItem("zorveus_auth_state", pkce.state);

  // 3. Build authorization URL
  const authUrl = ZorveusOAuth.getAuthorizationUrl({
    clientId: process.env.NEXT_PUBLIC_ZORVEUS_CLIENT_ID!,
    redirectUri: "http://localhost:3000/oauth/callback",
    state: pkce.state,
    codeChallenge: pkce.codeChallenge,
    scopes: ["inference:write", "models:*"],
  });

  // 4. Redirect browser to Zorveus authorization consent page
  window.location.href = authUrl;
}
```

### Step 2: Handle OAuth callback and token exchange

On your callback page (`http://localhost:3000/oauth/callback`):

```typescript
import { ZorveusOAuth, Zorveus } from "@zorveus/sdk";

export async function handleOAuthCallback() {
  const savedState = sessionStorage.getItem("zorveus_auth_state");
  const codeVerifier = sessionStorage.getItem("zorveus_code_verifier");

  // 1. Validate state CSRF token and extract authorization code
  const validation = ZorveusOAuth.validateCallback(window.location.href, savedState);

  if (!validation.valid || !validation.code || !codeVerifier) {
    throw new Error(`Authentication failed: ${validation.errorDescription}`);
  }

  // 2. Exchange authorization code for Bearer access token
  const tokenResponse = await ZorveusOAuth.exchangeToken({
    clientId: process.env.NEXT_PUBLIC_ZORVEUS_CLIENT_ID!,
    code: validation.code,
    codeVerifier,
    redirectUri: "http://localhost:3000/oauth/callback",
  });

  const accessToken = tokenResponse.access_token;
  sessionStorage.setItem("zorveus_access_token", accessToken);

  // 3. Initialize Zorveus client using the wallet access token
  const client = new Zorveus({ apiKey: accessToken });

  // 4. Test inference call
  const response = await client.chat.completions.create({
    model: "openai/gpt-4.1-mini",
    messages: [{ role: "user", content: "Hello from connected wallet!" }],
  });

  console.log(response.choices[0].message.content);
}
```
