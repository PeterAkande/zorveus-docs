When customers pay for AI usage (such as buying a $10 top-up pack or upgrading their plan), your payment provider handles the money, and your backend grants credits in Zorveus using a service key.

## 1. Issue an idempotent credit grant

Always provide an `external_reference` corresponding to your payment provider's transaction ID (e.g. Stripe PaymentIntent ID). If your webhook receives duplicate events or experiences a network retry, Zorveus safely deduplicates the request and returns the existing grant without adding duplicate balance.

<CodeGroup titles={["TypeScript", "Python", "cURL"]}>

```typescript
import { ZorveusServiceClient } from "@zorveus/sdk";

const service = new ZorveusServiceClient({ apiKey: "zrv_svc_your_service_key" });

const receipt = await service.productUsers.grantCreditByExternalId({
  appId: "app_123",
  externalUserId: "customer_42",
  amount: "10.00",
  externalReference: "stripe_pi_3N4v9B2eZvKYlo2C01234567",
  source: "purchase",
  reason: "100k token pack top-up",
});

console.log(`Granted: $${receipt.credit_grant.amount}`);
console.log(`New total available credits: $${receipt.credit_summary.available_credits}`);
```

```python
from zorveus import ZorveusServiceClient

service = ZorveusServiceClient(api_key="zrv_svc_your_service_key")

receipt = service.product_users.grant_credit_by_external_id(

```

## 2. Stripe webhook handler example

Here is how to handle Stripe's `payment_intent.succeeded` event in Next.js or Express:

<CodeGroup titles={["Next.js App Router (TypeScript)", "Express.js (TypeScript)"]}>

```typescript
// app/api/webhooks/stripe/route.ts
import { NextResponse } from "next/server";
import Stripe from "stripe";
import { ZorveusServiceClient } from "@zorveus/sdk";

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!);
const zorveus = new ZorveusServiceClient({ apiKey: process.env.ZORVEUS_SERVICE_KEY! });

export async function POST(req: Request) {
  const body = await req.text();
  const sig = req.headers.get("stripe-signature")!;

  let event: Stripe.Event;
  try {
    event = stripe.webhooks.constructEvent(body, sig, process.env.STRIPE_WEBHOOK_SECRET!);
  } catch (err: unknown) {
    return NextResponse.json({ error: "Invalid signature" }, { status: 400 });
  }

  if (event.type === "payment_intent.succeeded") {
    const paymentIntent = event.data.object as Stripe.PaymentIntent;
    const userId = paymentIntent.metadata.userId;
    const creditAmount = (paymentIntent.amount / 100).toFixed(2);

    await zorveus.productUsers.grantCreditByExternalId({
      appId: process.env.ZORVEUS_APP_ID!,
      externalUserId: userId,
      amount: creditAmount,
      externalReference: paymentIntent.id,
      source: "purchase",
      reason: `Credit pack top-up (${paymentIntent.id})`,
    });
  }

  return NextResponse.json({ received: true });
}
```

```typescript
// server.ts
import express from "express";
import Stripe from "stripe";
import { ZorveusServiceClient } from "@zorveus/sdk";

const app = express();
const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!);
const zorveus = new ZorveusServiceClient({ apiKey: process.env.ZORVEUS_SERVICE_KEY! });

app.post("/webhook/stripe", express.raw({ type: "application/json" }), async (req, res) => {
  const sig = req.headers["stripe-signature"] as string;

  let event: Stripe.Event;
  try {
    event = stripe.webhooks.constructEvent(req.body, sig, process.env.STRIPE_WEBHOOK_SECRET!);
  } catch (err) {
    return res.status(400).send("Webhook signature verification failed");
  }

  if (event.type === "payment_intent.succeeded") {
    const paymentIntent = event.data.object as Stripe.PaymentIntent;
    const userId = paymentIntent.metadata.userId;
    const creditAmount = (paymentIntent.amount / 100).toFixed(2);

    await zorveus.productUsers.grantCreditByExternalId({
      appId: process.env.ZORVEUS_APP_ID!,
      externalUserId: userId,
      amount: creditAmount,
      externalReference: paymentIntent.id,
      source: "purchase",
      reason: `Credit pack top-up (${paymentIntent.id})`,
    });
  }

  res.json({ received: true });
});
```

## 3. Verify an existing grant by reference

If you need to verify whether a customer was credited for a past transaction, query by reference instead of creating a new grant:

<CodeGroup titles={["TypeScript", "Python", "cURL"]}>

```typescript
const grant = await service.productUsers.getCreditGrantByExternalReference({
  appId: "app_123",
  externalUserId: "customer_42",
  externalReference: "stripe_pi_3N4v9B2eZvKYlo2C01234567",
});

console.log(`Grant status: ${grant.status}, remaining: $${grant.remaining_amount}`);
```

```python
grant = service.product_users.get_credit_grant_by_external_reference(
    app_id="app_123",
    external_user_id="customer_42",
    external_reference="stripe_pi_3N4v9B2eZvKYlo2C01234567",
)

print(f"Grant status: {grant.status}, remaining: ${grant.remaining_amount}")
```

```bash
curl --fail-with-body --get https://api.zorveus.com/product-users/by-external-id/credit-grants/by-reference \
	-H "Authorization: Bearer zrv_svc_your_service_key" \
	--data-urlencode "app_id=app_123" \
	--data-urlencode "external_user_id=customer_42" \
	--data-urlencode "external_reference=stripe_pi_3N4v9B2eZvKYlo2C01234567"
```
